\subsection{How to write a proof for points}
\R{generic scheme}{
\begin{rcp}
\item State what is given and what is to be shown ("Zu zeigen: \dots").
\item Reduce w.l.o.g.\ (\emph{oBdA}) --- e.g.\ replace $f$ by $-f$ to fix a sign.
\item Build the auxiliary object (a set, a helper function, a subsequence). \emph{This is where most points sit.}
\item Justify existence (completeness / Bolzano--Weierstrass / Weierstrass max).
\item Do the $\eps$-argument or the contradiction.
\item Close with the statement + $\square$.
\end{rcp}
}

\subsection{Intermediate value theorem \de{Zwischenwertsatz}}
\F{$f\in C([a,b])$, $c$ between $f(a),f(b)$ \ra\ $\exists x:f(x)=c$}{
\begin{rcp}
\item oBdA $f(a)\le c\le f(b)$ (else consider $-f$).
\item Define $X:=\{x\in[a,b]\mid f(x)\le c\}$.
\item $X\ne\emptyset$ ($a\in X$) and $X\subseteq[a,b]$ bounded \ra\ $x^*:=\sup X$ exists (completeness).
\item Choose $x_n\in X\cap[x^*-\frac1n,x^*]$ \ra\ $x_n\to x^*$ and $f(x_n)\le c$.
\item Continuity (sequential) \ra\ $f(x^*)=\lim f(x_n)\le c$.
\item Assume $f(x^*)<c$. Then $x^*<b$ (since $c\le f(b)$). Put $\eps:=c-f(x^*)>0$; continuity gives $\delta>0$ with $|y-x^*|<\delta\Rightarrow|f(y)-f(x^*)|<\eps$.
\item Hence $f(y)<f(x^*)+\eps=c$ for those $y$, i.e.\ $(x^*,x^*+\delta)\cap[a,b]\subseteq X$.
\item Contradiction to $x^*=\sup X$ \ra\ $f(x^*)=c$. $\square$
\end{rcp}
}

\subsection{Cauchy sequence converges \de{Vollst\"andigkeit von $\Rl$}}
\F{$(a_n)$ Cauchy in $\Rl$ \ra\ convergent}{
\begin{rcp}
\item Cauchy \ra\ \textbf{bounded} (take $\eps=1$, $N$; then $|a_n|\le\max\{|a_1|,\dots,|a_N|,|a_N|+1\}$).
\item Bolzano--Weierstrass \ra\ a convergent subsequence $a_{n_k}\to L$.
\item Cauchy: $\exists N$ with $|a_n-a_m|<\frac\eps2$ for all $n,m\ge N$.
\item Subsequence: $\exists N_1$ with $|a_{n_k}-L|<\frac\eps2$ for all $k\ge N_1$.
\item $N^*:=\max\{N,N_1\}$; using $n_k\ge k$ pick one index $n_{k}\ge N^*$.
\item Triangle inequality: $|a_n-L|\le|a_n-a_{n_k}|+|a_{n_k}-L|<\frac\eps2+\frac\eps2=\eps$ for all $n\ge N^*$ \ra\ $a_n\to L$. $\square$
\end{rcp}
}

\subsection{Rolle and the mean value theorem}
\F{Rolle: $f\in C[a,b]$, diff.\ on $(a,b)$, $f(a)=f(b)$ \ra\ $\exists\xi:f'(\xi)=0$}{
\begin{rcp}
\item $f$ continuous on the compact $[a,b]$ \ra\ attains $\max$ and $\min$ (Weierstrass).
\item If both are attained at the boundary then $f$ is constant \ra\ $f'\equiv0$, done.
\item Otherwise an extremum lies in the interior at some $\xi$ \ra\ Fermat \ra\ $f'(\xi)=0$. $\square$
\end{rcp}
}
\F{MVT (Mittelwertsatz): $\exists\xi\in(a,b):f'(\xi)=\frac{f(b)-f(a)}{b-a}$}{
\begin{rcp}
\item Auxiliary function $g(x):=f(x)-\dfrac{f(b)-f(a)}{b-a}(x-a)$.
\item $g$ is continuous on $[a,b]$, differentiable on $(a,b)$, and $g(a)=f(a)=g(b)$.
\item Rolle \ra\ $\exists\xi\in(a,b)$ with $0=g'(\xi)=f'(\xi)-\frac{f(b)-f(a)}{b-a}$. $\square$
\end{rcp}
}

\subsection{Further one-liners you may be asked}
\F{differentiable \ra\ continuous}{
$f(x)-f(x_0)=\frac{f(x)-f(x_0)}{x-x_0}(x-x_0)\to f'(x_0)\cdot0=0$. $\square$
}
\F{convergent \ra\ bounded}{
Take $\eps=1$, get $N$ with $|a_n-a|<1$ for $n\ge N$ \ra\ $|a_n|\le|a|+1$; the finitely many $a_1,\dots,a_{N-1}$ are bounded by their maximum. $\square$
}
\F{limit is unique}{
Assume $a_n\to a$ and $a_n\to b$, $a\ne b$. Take $\eps=\frac{|a-b|}2$; for large $n$: $|a-b|\le|a-a_n|+|a_n-b|<|a-b|$, contradiction. $\square$
}
\F{monotone + bounded \ra\ convergent}{
$(a_n)$ increasing, bounded \ra\ $s:=\sup\{a_n\}$ exists. Given $\eps>0$, $s-\eps$ is not an upper bound \ra\ $\exists N:a_N>s-\eps$; monotonicity \ra\ $s-\eps<a_N\le a_n\le s$ for $n\ge N$ \ra\ $|a_n-s|<\eps$. $\square$
}
\F{Bolzano--Weierstrass}{
Bisection: $(a_n)\subseteq[A,B]$. Halve the interval and keep a half containing infinitely many terms; repeat \ra\ nested intervals of length $\frac{B-A}{2^k}$. Pick $n_k$ increasing with $a_{n_k}$ in the $k$-th interval \ra\ the nested intervals shrink to a point $L$ and $a_{n_k}\to L$. $\square$
}
\F{uniform limit of continuous functions is continuous}{
$\frac\eps3$-argument: $|f(x)-f(x_0)|\le|f(x)-f_N(x)|+|f_N(x)-f_N(x_0)|+|f_N(x_0)-f(x_0)|$.
Choose $N$ with $\|f_N-f\|_\infty<\frac\eps3$ (uniformity: same $N$ for all $x$!), then $\delta$ from continuity of $f_N$ at $x_0$. $\square$
}
\F{absolute convergence \ra\ convergence}{
Cauchy criterion: $\left|\sum_{k=n}^{m}a_k\right|\le\sum_{k=n}^m|a_k|<\eps$ for $m\ge n\ge N$; the partial sums are Cauchy \ra\ converge. $\square$
}
\F{Fermat}{
$x_0$ interior local max \ra\ for $h>0$ small $\frac{f(x_0+h)-f(x_0)}{h}\le0$ and for $h<0$ it is $\ge0$; both one-sided limits equal $f'(x_0)$ \ra\ $f'(x_0)=0$. $\square$
}
\F{squeeze theorem}{
$b_n\le a_n\le c_n$, $b_n,c_n\to L$. Given $\eps$: for large $n$, $L-\eps<b_n\le a_n\le c_n<L+\eps$ \ra\ $|a_n-L|<\eps$. $\square$
}
\F{limit laws (sum, product)}{
$|(a_n+b_n)-(a+b)|\le|a_n-a|+|b_n-b|<\frac\eps2+\frac\eps2$.
Product: $|a_nb_n-ab|\le|a_n||b_n-b|+|b||a_n-a|$; $(a_n)$ is bounded by $C$ (convergent \ra\ bounded) \ra\ $\le C|b_n-b|+|b||a_n-a|<\eps$. $\square$
}
\F{product rule}{
$\frac{(fg)(x)-(fg)(x_0)}{x-x_0}=f(x)\frac{g(x)-g(x_0)}{x-x_0}+g(x_0)\frac{f(x)-f(x_0)}{x-x_0}$; let $x\to x_0$ and use $f(x)\to f(x_0)$ (continuity). $\square$
}
\F{chain rule}{
Put $\varphi(y):=\frac{g(y)-g(y_0)}{y-y_0}$ for $y\ne y_0$ and $\varphi(y_0):=g'(y_0)$; $\varphi$ is continuous at $y_0$ and $g(y)-g(y_0)=\varphi(y)(y-y_0)$. Insert $y=f(x)$, $y_0=f(x_0)$, divide by $x-x_0$, let $x\to x_0$. $\square$
}
\F{derivative of the inverse}{
$f^{-1}$ continuous, $f'(x_0)\ne0$: $\frac{f^{-1}(y)-f^{-1}(y_0)}{y-y_0}=\frac{x-x_0}{f(x)-f(x_0)}\to\frac1{f'(x_0)}$ as $y\to y_0$ (then $x\to x_0$). $\square$
}
\F{extreme value theorem (Weierstrass)}{
$f$ cont.\ on $[a,b]$. \textbf{(i) bounded:} else $\exists x_n$ with $|f(x_n)|\ge n$; Bolzano--Weierstrass \ra\ $x_{n_k}\to x^*\in[a,b]$, continuity \ra\ $f(x_{n_k})\to f(x^*)$ finite --- contradiction.
\textbf{(ii) attained:} $S:=\sup f$ exists; pick $x_n$ with $f(x_n)\to S$; a convergent subsequence $x_{n_k}\to x^*$ gives $f(x^*)=S$. $\square$
}
\F{continuous on a compact interval \ra\ uniformly continuous (Heine)}{
Suppose not: $\exists\eps_0>0$ and $x_n,y_n$ with $|x_n-y_n|<\frac1n$ but $|f(x_n)-f(y_n)|\ge\eps_0$. Bolzano--Weierstrass \ra\ $x_{n_k}\to x^*$, hence also $y_{n_k}\to x^*$. Continuity \ra\ $|f(x_{n_k})-f(y_{n_k})|\to0$ --- contradiction. $\square$
}
\F{fundamental theorem of calculus}{
$F(x)=\int_a^xf$. Then $\frac{F(x+h)-F(x)}{h}=\frac1h\int_x^{x+h}f=f(\xi_h)$ for some $\xi_h$ between $x$ and $x+h$ (MVT for integrals). $h\to0$ \ra\ $\xi_h\to x$ \ra\ continuity \ra\ $F'(x)=f(x)$. $\square$
}
\F{geometric series}{
$s_N=\sum_{k=0}^Nq^k$; $s_N-qs_N=1-q^{N+1}$ \ra\ $s_N=\frac{1-q^{N+1}}{1-q}$. $|q|<1$ \ra\ $q^{N+1}\to0$ \ra\ $s_N\to\frac1{1-q}$. $\square$
}
\F{root test}{
$\limsup\sqrt[n]{|a_n|}=q<1$: pick $q<r<1$; for $n\ge N$, $\sqrt[n]{|a_n|}\le r$ \ra\ $|a_n|\le r^n$ \ra\ comparison with the convergent geometric series. $q>1$ \ra\ $|a_n|\ge1$ infinitely often \ra\ $a_n\not\to0$. $\square$
}
\F{Cauchy--Schwarz}{
For all $t$: $0\le\|x+ty\|^2=\|x\|^2+2t\langle x,y\rangle+t^2\|y\|^2$. A quadratic in $t$ that is $\ge0$ has discriminant $\le0$: $4\langle x,y\rangle^2-4\|x\|^2\|y\|^2\le0$. $\square$
}
\F{a convergent sequence has exactly one accumulation point}{
$a_n\to a$: every subsequence also $\to a$ (its indices $n_k\ge k$), so $a$ is the only possible subsequential limit. $\square$
}
\F{$\left(1+\frac1n\right)^n$ converges}{
Increasing by AM--GM (or the binomial theorem term-by-term) and bounded above by $\sum_{k\ge0}\frac1{k!}<3$ \ra\ monotone $+$ bounded \ra\ converges; the limit is called $e$. $\square$
}
\F{monotone functions are Riemann integrable}{
$f$ increasing on $[a,b]$, equidistant partition with mesh $\frac{b-a}{n}$:
$U(f,P)-L(f,P)=\frac{b-a}{n}\sum_i\left(f(x_i)-f(x_{i-1})\right)=\frac{b-a}{n}\left(f(b)-f(a)\right)\to0$ (telescoping) \ra\ integrable. $\square$
}
\F{convergent \ra\ Cauchy}{
$|a_n-a_m|\le|a_n-a|+|a-a_m|<\frac\eps2+\frac\eps2$ for $n,m\ge N$. $\square$
}
\F{Archimedean property}{
If $\N$ were bounded above, $s:=\sup\N$ would exist; $s-1$ is not an upper bound \ra\ $\exists n>s-1$ \ra\ $n+1>s$, contradiction. \ra\ $\forall x\ \exists n>x$, hence $\frac1n\to0$. $\square$
}
\F{characterisation of the supremum}{
$s=\sup M$ $\iff$ (i) $x\le s\ \forall x\in M$ and (ii) $\forall\eps>0\ \exists x\in M:x>s-\eps$.
"$\Rightarrow$": if (ii) failed, $s-\eps$ would be a smaller upper bound. "$\Leftarrow$": (i) makes $s$ an upper bound, (ii) makes it the least. $\square$
}
\F{comparison test}{
$0\le a_n\le b_n$, $\sum b_n<\infty$: the partial sums of $\sum a_n$ are increasing and bounded by $\sum b_n$ \ra\ monotone $+$ bounded \ra\ converge. $\square$
}
\F{Leibniz criterion}{
$b_n\downarrow0$: $(s_{2n})$ is increasing, $(s_{2n+1})$ decreasing, $s_{2n}\le s_{2n+1}$, and $s_{2n+1}-s_{2n}=b_{2n+1}\to0$ \ra\ nested intervals \ra\ both converge to the same $s$. $\square$
}
\F{Cauchy--Hadamard}{
Apply the root test to $\sum a_n(x-x_0)^n$: $\limsup\sqrt[n]{|a_n||x-x_0|^n}=|x-x_0|\limsup\sqrt[n]{|a_n|}=\frac{|x-x_0|}{R}$, which is $<1$ exactly for $|x-x_0|<R$. $\square$
}
\F{uniform convergence allows swapping $\lim$ and $\int$}{
$\left|\int_a^bf_n-\int_a^bf\right|\le\int_a^b|f_n-f|\le(b-a)\|f_n-f\|_\infty\to0$. $\square$
}
\F{l'Hospital (idea)}{
$f(x_0)=g(x_0)=0$: Cauchy MVT gives $\frac{f(x)}{g(x)}=\frac{f(x)-f(x_0)}{g(x)-g(x_0)}=\frac{f'(\xi_x)}{g'(\xi_x)}$ with $\xi_x$ between $x_0$ and $x$; $x\to x_0$ forces $\xi_x\to x_0$. $\square$
}
\F{convex $\iff$ $f''\ge0$ (differentiable case)}{
$f$ convex $\iff$ $f'$ monotonically increasing (chord slopes increase) $\iff$ $f''\ge0$ (monotonicity criterion). $\square$
}
\F{contraction / uniqueness of a fixed point}{
$|f(x)-f(y)|\le q|x-y|$, $q<1$. Two fixed points \ra\ $|x^*-y^*|\le q|x^*-y^*|$ \ra\ $(1-q)|x^*-y^*|\le0$ \ra\ $x^*=y^*$. Existence: $x_{n+1}=f(x_n)$ is Cauchy since $|x_{n+1}-x_n|\le q^n|x_1-x_0|$. $\square$
}
