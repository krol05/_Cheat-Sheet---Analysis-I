%  Design rules, do not break them:
%   * every entry answers "I have THIS question -> do THIS"
%   * definitions survive only where an exercise needs them, and each
%     concept is defined exactly ONCE; everywhere else use \ref
%   * ONE continuous multicols{3} for the whole body: no \clearpage, no
%     nested multicols. That is what keeps the columns packed with no
%     seams and no orphan gaps.
%   * \raggedcolumns is load-bearing: multicol's default \flushcolumns
%     forces every column to \textheight by inflating the glue between
%     boxes, which scatters holes through the page (and \vbadness=10000
%     inside multicol hides the warnings).
%   * every box must stay well under 1/3 of a column: multicol cannot
%     split an unbreakable tcolorbox, so a tall box punches a hole.
%     Do NOT switch the boxes to `breakable' to fix that -- the break
%     decorations cost more space than the holes (tested: +2 pages).
%   * cross-references go through \label/\ref, never hard-coded numbers.
% =====================================================================

% ---------- Section banner ----------

\newcommand{\sectionbanner}[1]{%
  \par\tsNeed{6\baselineskip}\vspace{1.0ex}%
  \refstepcounter{section}%
  \label{sec:\arabic{section}}%
  \noindent{\color{secRule}\rule{\linewidth}{1.5pt}}\par\nointerlineskip
  \noindent\colorbox{secRule}{\parbox[c][3.0ex][c]{\dimexpr\linewidth-2\fboxsep\relax}{%
    \color{white}\bfseries\sffamily\fontsize{8.4pt}{10.0pt}\selectfont
    \colorbox{white}{\color{secRule}\,\thesection\,}\quad #1}}%
  \par\nointerlineskip\noindent{\color{secRule}\rule{\linewidth}{1.5pt}}%
  \par\vspace{0.6ex}%
}

% ---------- Which sections are live on this page? ----------
% Lo = last section that started strictly before this page (it carries over),
% Hi = last section that has started by this page. Live range = Lo..Hi.
% Built from \pageref, not TeX marks: multicol delivers marks out of page order.
\newcounter{tsSecLo}
\newcounter{tsSecHi}
\newcommand{\tsChkLo}[1]{\ifnum\getpagerefnumber{sec:#1}<\value{page}\relax\setcounter{tsSecLo}{#1}\fi}
\newcommand{\tsChkHi}[1]{\ifnum\getpagerefnumber{sec:#1}>\value{page}\relax\else\setcounter{tsSecHi}{#1}\fi}
\newcommand{\tsScanSections}{%
  \setcounter{tsSecLo}{0}\setcounter{tsSecHi}{0}%
  \tsChkLo{1}\tsChkHi{1}\tsChkLo{2}\tsChkHi{2}\tsChkLo{3}\tsChkHi{3}%
  \tsChkLo{4}\tsChkHi{4}\tsChkLo{5}\tsChkHi{5}\tsChkLo{6}\tsChkHi{6}%
  \tsChkLo{7}\tsChkHi{7}\tsChkLo{8}\tsChkHi{8}\tsChkLo{9}\tsChkHi{9}%
  \tsChkLo{10}\tsChkHi{10}\tsChkLo{11}\tsChkHi{11}\tsChkLo{12}\tsChkHi{12}%
  \tsChkLo{13}\tsChkHi{13}%
}

        font=\sffamily\bfseries\fontsize{6.6pt}{6.6pt}\selectfont]

% ---------- Top-of-page contents strip ----------
% "Which section is number 7, and what page is it on?" answered on every page without
% flipping back to p.1. Drawn as a TikZ overlay into the top margin, exactly like
% \tsTabStrip, so \textheight is untouched and nothing in the body moves.
% Sections live on this page are dark; the rest are grey.
\newcommand{\tsTocLive}[1]{%
  {\setlength{\fboxsep}{0.7pt}\colorbox{secRule}{\color{white}\bfseries#1}}%
  \,{\color{secRule}\bfseries\tsName{#1}}%
  \,{\color{secRule}\bfseries p\pageref{sec:#1}}}
\newcommand{\tsTocDim}[1]{%
  {\setlength{\fboxsep}{0.7pt}\colorbox{tocDim}{\color{white}\bfseries#1}}%
  \,{\color{tocDim}\bfseries\tsName{#1}}%
  \,{\color{tocDim}\bfseries p\pageref{sec:#1}}}
\newcommand{\tsTocEntry}[1]{%
  \ifnum#1<\value{tsSecLo}\relax\tsTocDim{#1}\else
  \ifnum#1>\value{tsSecHi}\relax\tsTocDim{#1}\else
  \tsTocLive{#1}\fi\fi}
\newcommand{\tsTocSep}{\hfill{\color{tocDim}\rule[-0.7pt]{0.5pt}{5.2pt}}\hfill}
\newcommand{\tsTocStrip}{%
  \tsScanSections
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north west, inner sep=0pt, outer sep=0pt]
      at ([xshift=0.40cm, yshift=-1.5pt]current page.north west)
      {\parbox{\textwidth}{\sffamily\fontsize{5.4pt}{6.2pt}\selectfont
        % without this the trailing \parfillskip eats a share of every \hfill and the
        % strip stops short of the right margin
        \setlength{\parfillskip}{0pt}%
        \tsTocEntry{1}\tsTocSep\tsTocEntry{2}\tsTocSep\tsTocEntry{3}\tsTocSep
        \tsTocEntry{4}\tsTocSep\tsTocEntry{5}\tsTocSep\tsTocEntry{6}\tsTocSep
        \tsTocEntry{7}\tsTocSep\tsTocEntry{8}\tsTocSep\tsTocEntry{9}\tsTocSep
        \tsTocEntry{10}\tsTocSep\tsTocEntry{11}\tsTocSep\tsTocEntry{12}\tsTocSep
        \tsTocEntry{13}}};
  \end{tikzpicture}%
}
